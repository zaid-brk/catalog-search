cmake_minimum_required(VERSION 3.24)
project(catalog_search VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
find_package(Boost 1.75 REQUIRED)
option(CATALOG_ASAN "Address and undefined behavior sanitizers" OFF)
option(CATALOG_TSAN "Thread sanitizer (separate build)" OFF)
option(CATALOG_MICROBENCHMARKS "Build Google Benchmark microbenchmarks" OFF)
if(CATALOG_ASAN AND CATALOG_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
if(CATALOG_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(CATALOG_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(catalog_core engine/src/index.cpp)
target_include_directories(catalog_core PUBLIC engine/include)
target_compile_options(catalog_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(catalog_core PUBLIC Threads::Threads)
add_library(catalog_json engine/src/json.cpp)
target_include_directories(catalog_json SYSTEM PUBLIC ${Boost_INCLUDE_DIRS})
target_link_libraries(catalog_json PUBLIC catalog_core)
add_executable(catalog engine/src/main.cpp)
target_link_libraries(catalog PRIVATE catalog_json)
add_executable(catalog-server engine/src/server.cpp)
target_link_libraries(catalog-server PRIVATE catalog_json)
add_executable(catalog-loadtest engine/src/loadtest.cpp)
target_link_libraries(catalog-loadtest PRIVATE catalog_json)
add_executable(catalog-allocation-bench benchmarks/search_allocations.cpp)
target_link_libraries(catalog-allocation-bench PRIVATE catalog_core)
include(CTest)
include(FetchContent)
if(BUILD_TESTING)
  if(GTEST_SOURCE_DIR)
    add_subdirectory(${GTEST_SOURCE_DIR} ${CMAKE_BINARY_DIR}/googletest EXCLUDE_FROM_ALL)
  else()
    find_package(GTest QUIET)
    if(NOT GTest_FOUND)
      FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
      FetchContent_MakeAvailable(googletest)
    endif()
  endif()
  add_executable(catalog-tests engine/tests/index_test.cpp)
  target_link_libraries(catalog-tests PRIVATE catalog_json GTest::gtest_main)
  if(APPLE)
    # Some installed GTest packages use @rpath install names without exporting
    # their library directory as a runtime search path. Discovery runs this
    # binary during the build, so it must be able to load both libraries then.
    set_property(TARGET catalog-tests APPEND PROPERTY BUILD_RPATH
      "$<TARGET_FILE_DIR:GTest::gtest_main>"
      "$<TARGET_FILE_DIR:GTest::gtest>")
  endif()
  include(GoogleTest)
  gtest_discover_tests(catalog-tests)
endif()
if(CATALOG_MICROBENCHMARKS)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark URL https://github.com/google/benchmark/archive/refs/tags/v1.9.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(benchmark)
  add_executable(catalog-microbench benchmarks/microbench.cpp)
  target_link_libraries(catalog-microbench PRIVATE catalog_core benchmark::benchmark_main)
endif()
